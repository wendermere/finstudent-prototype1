  <script>
    const STORAGE_KEY = 'finstudent_prototype_state_v2';
    const USER_KEY = 'finstudent_prototype_user_v2';
    let authMode = 'login';

    function getDefaultState() {
      return {
        allowance: 1500,
        cycle: 'Weekly',
        daysLeft: 4,
        dailySeries: [185, 220, 160, 200, 0, 0, 0],
        categories: [
          { name: 'Food & Drinks', budget: 450, spent: 375 },
          { name: 'Transportation', budget: 300, spent: 180 },
          { name: 'School & Printing', budget: 450, spent: 120 },
          { name: 'Personal & Leisure', budget: 300, spent: 90 }
        ],
        expenses: [
          { id: 1, category: 'Food & Drinks', amount: 145, note: 'Campus lunch & iced drink', date: 'Today' },
          { id: 2, category: 'Transportation', amount: 60, note: 'Jeepney & tricycle commute', date: 'Today' },
          { id: 3, category: 'School & Printing', amount: 120, note: 'Thesis draft printouts', date: 'Yesterday' }
        ],
        savingsGoals: [
          { name: 'Emergency Buffer', saved: 850, target: 2000 },
          { name: 'Capstone & Defense Fund', saved: 600, target: 1500 }
        ]
      };
    }

    let state = JSON.parse(localStorage.getItem(STORAGE_KEY)) || getDefaultState();
    let currentUser = JSON.parse(localStorage.getItem(USER_KEY)) || null;

    let dailyChartInstance = null;
    let budgetChartInstance = null;

    const peso = (num) => '₱' + Number(num || 0).toLocaleString('en-PH', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
    const persistState = () => localStorage.setItem(STORAGE_KEY, JSON.stringify(state));

    function setAuthMode(mode) {
      authMode = mode;
      const isReg = mode === 'register';
      document.getElementById('registerNameGroup').classList.toggle('hidden', !isReg);
      document.getElementById('registerCourseGroup').classList.toggle('hidden', !isReg);
      document.getElementById('authSubmitBtn').textContent = isReg ? 'Register & Open Dashboard' : 'Sign In to Dashboard';

      document.getElementById('tabLoginBtn').className = isReg
        ? 'py-2 rounded-lg text-slate-500 transition'
        : 'py-2 rounded-lg bg-white text-emerald-700 shadow-sm transition';
      document.getElementById('tabRegisterBtn').className = isReg
        ? 'py-2 rounded-lg bg-white text-emerald-700 shadow-sm transition'
        : 'py-2 rounded-lg text-slate-500 transition';
    }

    function handleAuthSubmit(e) {
      e.preventDefault();
      const email = document.getElementById('authEmail').value.trim();
      const password = document.getElementById('authPassword').value.trim();
      const errBox = document.getElementById('authError');

      if (!email || password.length < 4) {
        errBox.textContent = 'Please enter a valid student email and password (min 4 characters).';
        errBox.classList.remove('hidden');
        return;
      }
      errBox.classList.add('hidden');

      let name = 'Wendermere Aguirre';
      let course = 'BSCS - 3rd Year';
      if (authMode === 'register') {
        name = document.getElementById('regName').value.trim() || email.split('@')[0];
        course = document.getElementById('regCourse').value.trim() || 'Tertiary Student';
      }

      currentUser = { name, email, course };
      localStorage.setItem(USER_KEY, JSON.stringify(currentUser));
      updateSessionUI();
    }

    function loginAsDemo() {
      currentUser = {
        name: 'Wendermere Aguirre',
        email: 'student@ortigas-cainta.sti.edu.ph',
        course: 'BSCS • STI Ortigas-Cainta'
      };
      localStorage.setItem(USER_KEY, JSON.stringify(currentUser));
      updateSessionUI();
    }

    function logoutUser() {
      currentUser = null;
      localStorage.removeItem(USER_KEY);
      updateSessionUI();
    }

    function updateSessionUI() {
      const loggedIn = Boolean(currentUser);
      document.getElementById('authScreen').classList.toggle('hidden', loggedIn);
      document.getElementById('appScreen').classList.toggle('hidden', !loggedIn);

      if (loggedIn) {
        document.getElementById('welcomeStudentText').textContent = `${currentUser.name} • ${currentUser.course}`;
        renderAll();
      }
    }

    function openModal(id) { document.getElementById(id).classList.remove('hidden'); }
    function closeModal(id) { document.getElementById(id).classList.add('hidden'); }

    function switchTab(tab) {
      ['dashboard', 'analytics', 'savings'].forEach(t => {
        document.getElementById('view-' + t).classList.toggle('hidden', t !== tab);
        const btn = document.getElementById('nav-' + t);
        btn.classList.toggle('text-emerald-600', t === tab);
        btn.classList.toggle('text-slate-400', t !== tab);
      });
      if (tab === 'analytics') renderCharts();
    }

    function saveAllowance() {
      const amt = parseFloat(document.getElementById('allowanceInput').value) || 0;
      const cycle = document.getElementById('cycleInput').value;
      const days = Math.max(1, parseInt(document.getElementById('daysLeftInput').value, 10) || 1);
      const preset = document.getElementById('rulePreset').value;

      state.allowance = amt;
      state.cycle = cycle;
      state.daysLeft = days;

      const spentMap = Object.fromEntries(state.categories.map(c => [c.name, c.spent]));
      const splits = preset === 'commuteHeavy'
        ? [
            { name: 'Food & Drinks', ratio: 0.40 },
            { name: 'Transportation', ratio: 0.35 },
            { name: 'School & Printing', ratio: 0.15 },
            { name: 'Personal & Leisure', ratio: 0.10 }
          ]
        : [
            { name: 'Food & Drinks', ratio: 0.30 },
            { name: 'Transportation', ratio: 0.20 },
            { name: 'School & Printing', ratio: 0.30 },
            { name: 'Personal & Leisure', ratio: 0.20 }
          ];

      state.categories = splits.map(s => ({
        name: s.name,
        budget: Math.round(amt * s.ratio),
        spent: spentMap[s.name] || 0
      }));

      persistState();
      closeModal('allowanceModal');
      renderAll();
    }

    function addExpense() {
      const amount = parseFloat(document.getElementById('expenseAmount').value);
      const categoryName = document.getElementById('expenseCategory').value;
      const note = document.getElementById('expenseNote').value.trim() || 'Daily expense';

      if (!amount || amount <= 0) return;

      const cat = state.categories.find(c => c.name === categoryName);
      if (cat) cat.spent += amount;

      state.dailySeries[4] = (state.dailySeries[4] || 0) + amount;

      state.expenses.unshift({
        id: Date.now(),
        category: categoryName,
        amount,
        note,
        date: 'Just now'
      });

      document.getElementById('expenseAmount').value = '';
      document.getElementById('expenseNote').value = '';
      persistState();
      closeModal('expenseModal');
      renderAll();
    }

    function addSavingsBoost() {
      if (state.savingsGoals.length > 0) {
        state.savingsGoals[0].saved += 100;
        persistState();
        renderSavings();
      }
    }

    function resetDemoData() {
      state = getDefaultState();
      persistState();
      renderAll();
    }

    function renderAll() {
      const totalSpent = state.categories.reduce((sum, c) => sum + c.spent, 0);
      const remaining = Math.max(0, state.allowance - totalSpent);
      const safeDaily = remaining / Math.max(1, state.daysLeft);

      document.getElementById('remainingTotal').textContent = peso(remaining);
      document.getElementById('totalAllocatedText').textContent = peso(state.allowance);
      document.getElementById('safeDailySpend').textContent = peso(safeDaily) + '/day';
      document.getElementById('cycleBadge').textContent = `${state.cycle} • ${state.daysLeft} days left`;

      const alertBox = document.getElementById('alertContainer');
      alertBox.innerHTML = '';
      state.categories.forEach(c => {
        if (c.budget <= 0) return;
        const pct = (c.spent / c.budget) * 100;
        if (pct >= 100) {
          alertBox.innerHTML += `
            <div class="bg-red-50 border border-red-200 text-red-800 text-xs px-3.5 py-2.5 rounded-xl font-medium">
              <strong>Limit Exceeded:</strong> ${c.name} is at ${pct.toFixed(0)}% (${peso(c.spent)} / ${peso(c.budget)}).
            </div>`;
        } else if (pct >= 80) {
          alertBox.innerHTML += `
            <div class="bg-amber-50 border border-amber-200 text-amber-800 text-xs px-3.5 py-2.5 rounded-xl font-medium">
              <strong>80% Threshold Alert:</strong> ${c.name} has reached ${pct.toFixed(0)}% of its budget.
            </div>`;
        }
      });

      const catContainer = document.getElementById('categoryCards');
      catContainer.innerHTML = state.categories.map(c => {
        const pct = c.budget > 0 ? Math.min(100, (c.spent / c.budget) * 100) : 0;
        const rawPct = c.budget > 0 ? (c.spent / c.budget) * 100 : 0;
        const barColor = rawPct >= 100 ? 'bg-red-500' : rawPct >= 80 ? 'bg-amber-500' : 'bg-emerald-500';
        return `
          <div class="bg-white p-3.5 rounded-2xl border border-slate-200">
            <div class="flex justify-between text-xs font-semibold mb-1.5">
              <span class="text-slate-800">${c.name}</span>
              <span class="tabular-nums text-slate-600">${peso(c.spent)} <span class="text-slate-400">/ ${peso(c.budget)}</span></span>
            </div>
            <div class="w-full h-2 bg-slate-100 rounded-full overflow-hidden">
              <div class="${barColor} h-full rounded-full transition-all" style="width:${pct}%"></div>
            </div>
          </div>`;
      }).join('');

      document.getElementById('txCount').textContent = `${state.expenses.length} items`;
      document.getElementById('recentExpenses').innerHTML = state.expenses.slice(0, 5).map(e => `
        <div class="p-3 flex items-center justify-between text-xs">
          <div>
            <p class="font-semibold text-slate-800">${e.note}</p>
            <p class="text-slate-400">${e.category} • ${e.date}</p>
          </div>
          <span class="font-bold text-slate-900 tabular-nums">-${peso(e.amount)}</span>
        </div>`).join('');

      renderSavings();
      if (!document.getElementById('view-analytics').classList.contains('hidden')) {
        renderCharts();
      }
    }

    function renderSavings() {
      document.getElementById('savingsCards').innerHTML = state.savingsGoals.map(g => {
        const pct = g.target > 0 ? Math.min(100, (g.saved / g.target) * 100) : 0;
        return `
          <div class="bg-white p-4 rounded-2xl border border-slate-200 space-y-2">

    function renderCharts() {
      const dailyCtx = document.getElementById('dailyChart').getContext('2d');
      const budgetCtx = document.getElementById('budgetChart').getContext('2d');

      if (dailyChartInstance) dailyChartInstance.destroy();
      if (budgetChartInstance) budgetChartInstance.destroy();

      dailyChartInstance = new Chart(dailyCtx, {
        type: 'line',
        data: {
          labels: ['Mon', 'Tue', 'Wed', 'Thu', 'Fri (Today)', 'Sat', 'Sun'],
          datasets: [{
            label: 'Daily Spend (₱)',
            data: state.dailySeries,
            borderColor: '#059669',
            backgroundColor: 'rgba(5, 150, 105, 0.12)',
            fill: true,
            tension: 0.35
          }]
        },
        options: { responsive: true, plugins: { legend: { display: false } } }
      });

      budgetChartInstance = new Chart(budgetCtx, {
        type: 'bar',
        data: {
          labels: state.categories.map(c => c.name.split(' ')[0]),
          datasets: [
            { label: 'Planned (₱)', data: state.categories.map(c => c.budget), backgroundColor: '#cbd5e1' },
            { label: 'Actual (₱)', data: state.categories.map(c => c.spent), backgroundColor: '#059669' }
          ]
        },
        options: { responsive: true }
      });
    }
